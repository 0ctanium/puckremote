---
title: Programmatic API
description: Use build, publish, pull and activate from Node, with any artifact store.
---

```ts
import { activate, build, publish, pull } from '@puck-remote/cli';

await pull({ cwd: 'my-theme', artifacts: myS3Store });
await build({ cwd: 'my-theme', outDir: 'dist', quiet: true });
const { id } = await publish({ distDir: 'my-theme/dist', artifacts: myS3Store }); // any ArtifactStore
await activate(myS3Store, id);
```

## `build(options)`

<TypeTable
  type={{
    cwd: { description: "Theme folder.", type: 'string', required: true },
    outDir: { description: "Output folder, relative to cwd.", type: 'string', default: "'dist'" },
    quiet: { description: "No console output.", type: 'boolean', default: 'false' },
  }}
/>

Returns `{ manifest, outDir }`. Throws `BuildError` with a path-prefixed message.
`BROWSER_EXTERNALS` lists the imports `bundle.browser.js` leaves to the editor.

## `publish(options)`

<TypeTable
  type={{
    distDir: { description: "A build output folder.", type: 'string', required: true },
    artifacts: { description: "An ArtifactStore, or a directory path (file-system store).", type: 'ArtifactStore | string', required: true },
    quiet: { description: "No console output.", type: 'boolean', default: 'false' },
  }}
/>

Returns `{ id }`, the id the store gave the artifact, which is now current.

## `pull(options)`

<TypeTable
  type={{
    cwd: { description: "Theme folder; pages go to <cwd>/pages.", type: 'string', required: true },
    artifacts: { description: "An ArtifactStore, or a directory path.", type: 'ArtifactStore | string', required: true },
    artifact: { description: "An artifact id.", type: 'string', default: 'the current artifact' },
    quiet: { description: "No console output.", type: 'boolean', default: 'false' },
  }}
/>

Returns `{ id, pages }` (the files written). Throws when nothing was published.

## `activate(artifacts, id)`

<TypeTable
  type={{
    artifacts: { description: "An ArtifactStore, or a directory path.", type: 'ArtifactStore | string', required: true },
    id: { description: "An existing artifact id.", type: 'string', required: true },
  }}
/>
