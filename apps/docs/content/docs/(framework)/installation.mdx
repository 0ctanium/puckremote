## Requirements

- **Node.js 24+** on Linux x64/arm64, or **native arm64 Node 26** on Apple Silicon
  (`node -p process.arch` must print `arm64`; isolated-vm has no darwin-x64 prebuild).
- Run Node with `NODE_OPTIONS=--no-node-snapshot` (required by isolated-vm).
- Next.js 16 (App Router), React 19.

<Steps>
<Step>

### Install the packages

```npm
npm install @puck-remote/core @puck-remote/next @puckeditor/core
```

Pick adapters for where your artifacts, pages and data live. The file-based ones are good to start:

```npm
npm install @puck-remote/artifacts-fs @puck-remote/pages-fs
```

</Step>
<Step>

### Create the config

The config is the **only file that knows your backend**. Every option is described in
[Configuration](/docs/configuration).

```ts title="puck-remote.config.ts"
import path from 'node:path';
import { defineConfig, devAllowAll, sharedSecretAuth } from '@puck-remote/core/config';
import { fsArtifactStore } from '@puck-remote/artifacts-fs';
import { fsPageStore } from '@puck-remote/pages-fs';
import { myCms } from './src/cms'; // your DataSource, see Core › Adapters › Data source

const root = process.cwd(); // import.meta.dirname is undefined in Next server bundles
const prod = process.env.NODE_ENV === 'production';

export default defineConfig({
  artifacts: fsArtifactStore({ dir: path.join(root, 'artifacts') }),
  pages: fsPageStore({ dir: path.join(root, 'data/pages') }),
  source: myCms(),
  auth: prod ? sharedSecretAuth({ secret: process.env.PUCK_REMOTE_ADMIN_TOKEN }) : devAllowAll(),
  origins: {
    site: ['https://www.example.com'],
    editor: ['https://admin.example.com'],
  },
  site: { name: 'My site', locale: 'en' },
});
```

</Step>
<Step>

### Add the app files

Five small files wire the framework into Next.js. They are explained in [Next.js setup](/docs/next-js).

```files
my-app
├── puck-remote.config.ts
├── next.config.ts
└── src
    ├── puck-remote.ts
    ├── proxy.ts
    └── app
        ├── [[...path]]
        │   └── page.tsx
        ├── editor
        │   └── [[...path]]
        │       └── page.tsx
        ├── api
        │   └── [[...path]]
        │       └── route.ts
        └── theme
            └── [[...path]]
                └── route.ts
```

```ts tab="src/puck-remote.ts"
import { createPuckRemote } from '@puck-remote/next';
import config from '../puck-remote.config';

export const remote = createPuckRemote(config);
```

```ts tab="src/proxy.ts"
import { createProxy } from '@puck-remote/next/proxy';
import remoteConfig from '../puck-remote.config';

export const proxy = createProxy(remoteConfig);
export const config = { matcher: ['/((?!_next/|favicon\\.ico).*)'] };
```

```tsx tab="app/[[...path]]/page.tsx"
import { PuckRemotePage, type PageProps } from '@puck-remote/next';
import { remote } from '@/puck-remote';

export const dynamic = 'force-dynamic';
export const generateMetadata = remote.generateMetadata;

export default async function Page(props: PageProps) {
  return <PuckRemotePage page={await remote.loadPage(props)} />;
}
```

```tsx tab="app/editor/[[...path]]/page.tsx"
import { EditorClient } from '@puck-remote/next';
import { remote } from '@/puck-remote';

export default async function Editor(props: { params: Promise<{ path?: string[] }> }) {
  return <EditorClient {...await remote.loadEditor(props)} />;
}
```

```ts tab="app/api/[[...path]]/route.ts"
import { remote } from '@/puck-remote';

export const { GET, POST } = remote.api;
```

```ts tab="app/theme/[[...path]]/route.ts"
import { remote } from '@/puck-remote';

export const { GET, HEAD } = remote.theme;
```

</Step>
<Step>

### Wrap the Next config

```ts title="next.config.ts"
import { withPuckRemote } from '@puck-remote/next/config';

export default withPuckRemote({
  allowedDevOrigins: ['*.localhost'], // the editor runs on editor.localhost in development
});
```

</Step>
<Step>

### Publish a theme

Create a theme (see [Build a theme](/docs/guides/theme-development)), then build and publish it
into the same artifact directory:

```bash
npx puck-remote build
```

```bash
npx puck-remote publish --artifacts ../my-app/artifacts
```

</Step>
<Step>

### Run it

```bash
NODE_OPTIONS=--no-node-snapshot npx next dev
```

Open the site on `http://localhost:3000` and the editor on `http://editor.localhost:3000/editor`.
In development without `origins`, every surface answers on every hostname; in production the
site and the editor **must** use distinct origins (see [Deploying safely](/docs/guides/deploying)).

</Step>
</Steps>
