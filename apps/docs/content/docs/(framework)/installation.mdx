## Requirements

- **Node.js 24+** on Linux x64/arm64, or **native arm64 Node 26** on Apple Silicon
  (`node -p process.arch` must print `arm64`; isolated-vm has no darwin-x64 prebuild).
- Run Node with `NODE_OPTIONS=--no-node-snapshot` (required by isolated-vm).
- Next.js 16 (App Router), React 19.

<Steps>
<Step>

### Install the packages

```npm
npm install @puck-remote/core @puck-remote/next @puck-remote/editor @puckeditor/core
```

Pick an artifact store (themes: code and pages) and a data source. The file-based store is good to
start:

```npm
npm install @puck-remote/artifacts-fs
```

</Step>
<Step>

### Create the config

The config is the **only file that knows your backend**. Every option is described in
[Configuration](/docs/configuration).

```ts title="puck-remote.config.ts"
import path from 'node:path';
import { defineConfig } from '@puck-remote/core/config';
import { fsArtifactStore } from '@puck-remote/artifacts-fs';
import { myCms } from './src/cms'; // your DataSource, see Core › Adapters › Data source

const root = process.cwd(); // import.meta.dirname is undefined in Next server bundles

export default defineConfig({
  artifacts: fsArtifactStore({ dir: path.join(root, 'artifacts') }),
  source: myCms(),
  origins: {
    admin: ['https://admin.example.com'], // your pages that embed the editor
    editor: 'https://editor.example.net', // where the editor is served (another site)
  },
  site: { name: 'My site', locale: 'en' },
});
```

</Step>
<Step>

### Add the app files

A few small files wire the framework into Next.js. They are explained in [Next.js setup](/docs/next-js).

```files
my-app
├── puck-remote.config.ts
├── next.config.ts
└── src
    ├── puck-remote.ts
    ├── proxy.ts
    └── app
        ├── [[...path]]
        │   └── page.tsx
        ├── admin
        │   └── [[...path]]
        │       ├── page.tsx
        │       └── AdminEditor.tsx
        ├── api
        │   └── editor-rpc
        │       └── route.ts
        ├── editor
        │   └── [[...path]]
        │       └── route.ts
        └── theme
            └── [[...path]]
                └── route.ts
```

```ts tab="src/puck-remote.ts"
import { createPuckRemote } from '@puck-remote/next';
import config from '../puck-remote.config';

export const remote = createPuckRemote(config);
```

```ts tab="src/proxy.ts"
import { createProxy } from '@puck-remote/next/proxy';
import remoteConfig from '../puck-remote.config';

export const proxy = createProxy(remoteConfig);
export const config = { matcher: ['/((?!_next/|favicon\\.ico).*)'] };
```

```tsx tab="app/[[...path]]/page.tsx"
import { PuckRemotePage, type PageProps } from '@puck-remote/next';
import { remote } from '@/puck-remote';

export const dynamic = 'force-dynamic';
export const generateMetadata = remote.generateMetadata;

export default async function Page(props: PageProps) {
  return <PuckRemotePage page={await remote.loadPage(props)} />;
}
```

```tsx tab="app/admin/[[...path]]/page.tsx"
import { remote } from '@/puck-remote';
import { AdminEditor } from './AdminEditor';

export const dynamic = 'force-dynamic';

export default async function Admin(props: { params: Promise<{ path?: string[] }> }) {
  // Check your own admin session here first.
  return <AdminEditor payload={await remote.loadEditor(props)} />;
}
```

```tsx tab="app/admin/[[...path]]/AdminEditor.tsx"
'use client';
import { PuckEditorFrame } from '@puck-remote/editor/frame';
import type { EditorPayload } from '@puck-remote/next';

const call = async (method: string, params: unknown) =>
  (await (await fetch('/api/editor-rpc', { method: 'POST', body: JSON.stringify({ method, params }) })).json()).value;

export function AdminEditor({ payload }: { payload: EditorPayload }) {
  return (
    <PuckEditorFrame
      editorUrl="https://editor.example.net/"
      editorOrigin="https://editor.example.net"
      payload={payload}
      rpc={{ resolveData: (p) => call('resolveData', { ...(p as object), slug: payload.slug }) }}
      onChange={(data) => {/* keep it for your Publish button */}}
    />
  );
}
```

```ts tab="app/api/editor-rpc/route.ts"
import { remote } from '@/puck-remote';

export const { POST } = remote.createEditorRpcRoute({
  async resolveData(params, request) {
    // Check your own admin session on `request` here.
    const { slug, block, props } = params as { slug: string; block: string; props: Record<string, unknown> };
    return remote.core.resolveBlockData(slug, block, props);
  },
});
```

```ts tab="app/editor/[[...path]]/route.ts"
import { remote } from '@/puck-remote';

// Served only on origins.editor (the proxy rewrites that origin here).
export const { GET, HEAD } = remote.editor;
```

```ts tab="app/theme/[[...path]]/route.ts"
import { remote } from '@/puck-remote';

export const { GET, HEAD } = remote.theme;
```

</Step>
<Step>

### Wrap the Next config

```ts title="next.config.ts"
import { withPuckRemote } from '@puck-remote/next/config';

export default withPuckRemote({
  allowedDevOrigins: ['*.localhost'], // admin pages run on admin.localhost in development
});
```

</Step>
<Step>

### Publish a theme

Create a theme (see [Build a theme](/docs/guides/theme-development)), then build and publish it
into the same artifact directory:

```bash
npx puck-remote build
```

```bash
npx puck-remote publish --artifacts ../my-app/artifacts
```

</Step>
<Step>

### Point the editor's hostname at the app

The editor ships built in `@puck-remote/editor`; the `app/editor` route above serves it on
`origins.editor`. Point that hostname (another site) at this deployment, or serve the editor
elsewhere with the `puck-remote-editor` bin. See [Serving the editor](/docs/guides/editor-app).

</Step>
<Step>

### Run it

```bash
NODE_OPTIONS=--no-node-snapshot npx next dev
```

Open the site on `http://localhost:3000` and the admin page on `http://admin.localhost:3000/admin`
(with `origins.admin` set to that origin in development). The editor must run on a different
origin from the admin pages (see [Origins](/docs/concepts/origins)).

</Step>
</Steps>
