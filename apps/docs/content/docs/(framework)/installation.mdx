## Requirements

- **Node.js 26+** on Linux x64/arm64, or **native arm64** on Apple Silicon (`node -p process.arch`
  must print `arm64`; isolated-vm has no darwin-x64 prebuild). Node 24 is not supported: its
  permission model does not block the render workers' network ([D-0402](/docs/internal/decisions#D-0402)).
- Run Node with `NODE_OPTIONS=--no-node-snapshot` (required by isolated-vm).
- Next.js 16 (App Router), React 19.

<Steps>
<Step>

### Install the packages

```npm
npm install @puck-remote/core @puck-remote/next @puck-remote/editor @puckeditor/core
```

Pick an artifact store (the site's themes and their files: code and templates) and a data source. The file-based store is good to
start:

```npm
npm install @puck-remote/artifacts-fs
```

</Step>
<Step>

### Create the config

The config is the **only file that knows your backend**. Every option is described in
[Configuration](/docs/configuration).

```ts title="puck-remote.config.ts"
import path from 'node:path';
import { defineConfig } from '@puck-remote/core/config';
import { fsArtifactStore } from '@puck-remote/artifacts-fs';
import { myCms } from './src/cms'; // your DataSource, see Core › Adapters › Data source

const root = process.cwd(); // import.meta.dirname is undefined in Next server bundles

export default defineConfig({
  artifacts: fsArtifactStore({ dir: path.join(root, 'artifacts') }),
  source: myCms(),
  origins: {
    host: ['https://admin.example.com'], // your admin pages that embed the editor
    editor: 'https://editor.example.net', // where the editor is served (another site)
  },
  site: { name: 'My site', locale: 'en' },
});
```

</Step>
<Step>

### Add the app files

A few small files wire the framework into Next.js. They are explained in [Next.js setup](/docs/next-js).

```files
my-app
├── puck-remote.config.ts
├── next.config.ts
└── src
    ├── puck-remote.ts
    ├── proxy.ts
    └── app
        ├── [[...path]]
        │   └── page.tsx
        ├── admin
        │   └── editor
        │       ├── page.tsx
        │       ├── ClientEditor.tsx
        │       └── actions.ts
        ├── api
        │   └── themes
        │       └── [[...path]]
        │           └── route.ts
        ├── cdn
        │   └── [[...path]]
        │       └── route.ts
        └── theme
            ├── editor
            │   ├── page.tsx
            │   └── EditorPage.tsx
            └── preview
                └── route.ts
```

```ts tab="src/puck-remote.ts"
import { createPuckRemote } from '@puck-remote/next';
import config from '../puck-remote.config';

export const remote = createPuckRemote(config);
```

```ts tab="src/proxy.ts"
import { createProxy } from '@puck-remote/next/proxy';
import remoteConfig from '../puck-remote.config';

// Without `template`, no cache headers: see the Next.js setup for a path → template mapping.
export const proxy = createProxy(remoteConfig);
export const config = { matcher: ['/((?!_next/|favicon\\.ico).*)'] };
```

```tsx tab="app/[[...path]]/page.tsx"
import { PuckRemoteTemplate, type RouteProps } from '@puck-remote/next';
import { remote } from '@/puck-remote';

export const dynamic = 'force-dynamic';

// Your routing: here, the template named like the path (/ → home).
export default async function Page({ params, searchParams }: RouteProps) {
  const name = (await params).path?.join('/') || 'home';
  return <PuckRemoteTemplate template={await remote.loadTemplate(name, { searchParams })} />;
}
```

```tsx tab="app/admin/editor/page.tsx"
import { remote } from '@/puck-remote';
import { ClientEditor } from './ClientEditor';

export const dynamic = 'force-dynamic';

// /editor?previewPath=/blog/post opens the template that page renders (none → home).
export default async function Admin(props: { searchParams: Promise<{ previewPath?: string }> }) {
  // Check your own admin session here first. loadEditor 404s outside origins.host.
  const name = (await props.searchParams).previewPath?.replace(/^\/+|\/+$/g, '') || 'home';
  return <ClientEditor payload={await remote.loadEditor(name)} />;
}
```

```tsx tab="app/admin/editor/ClientEditor.tsx"
'use client';
import { PuckEditorFrame } from '@puck-remote/editor/frame';
import type { EditorPayload } from '@puck-remote/next';
import { resolveData } from './actions';

// This page's Puck holds the data, the history and the fields; the frame shows the canvas.
// Add your header (undo/redo, Publish) as children: see Customizing the editor.
export function ClientEditor({ payload }: { payload: EditorPayload }) {
  return (
    <div style={{ height: '100vh' }}>
      <PuckEditorFrame
        editorUrl="https://editor.example.net/"
        payload={payload}
        resolveData={async (block, props, template) => (await resolveData({ template: template.name, params: template.params, block, props })).data}
      />
    </div>
  );
}
```

```ts tab="app/admin/editor/actions.ts"
'use server';
import { remote } from '@/puck-remote';

export async function resolveData(p: { template: string; params: Record<string, string>; block: string; props: Record<string, unknown> }) {
  // Check your own admin session here.
  return remote.core.resolveBlockData(p.template, p.block, p.props, { params: p.params });
}
```

```tsx tab="app/theme/editor/page.tsx"
import { remote } from '@/puck-remote';
import { EditorPage } from './EditorPage';

export const dynamic = 'force-dynamic';

// Served only on origins.editor (the proxy rewrites that origin here).
export default async function Editor() {
  const { allowedParents } = await remote.loadEditorPage();
  return <div style={{ height: '100vh' }}><EditorPage allowedParents={allowedParents} /></div>;
}
```

```tsx tab="app/theme/editor/EditorPage.tsx"
'use client';
import '@puckeditor/core/no-external.css';
import { PuckRemoteEditor } from '@puck-remote/editor/react';

export function EditorPage({ allowedParents }: { allowedParents: string[] }) {
  return <PuckRemoteEditor allowedParents={allowedParents} />;
}
```

```ts tab="app/cdn/[[...path]]/route.ts"
import { remote } from '@/puck-remote';

export const { GET, HEAD } = remote.createThemeHandler();
```

```ts tab="app/api/themes/[[...path]]/route.ts"
import { timingSafeEqual } from 'node:crypto';
import { remote } from '@/puck-remote';

// The artifacts API the CLI publishes through. Your auth: here one deploy token.
export const { GET, POST, PATCH, DELETE } = remote.createArtifactsHandler({
  authorize(request) {
    const expected = Buffer.from(`Bearer ${process.env.DEPLOY_TOKEN ?? ''}`);
    const given = Buffer.from(request.headers.get('authorization') ?? '');
    return !!process.env.DEPLOY_TOKEN && given.length === expected.length && timingSafeEqual(given, expected);
  },
});
```

```ts tab="app/theme/preview/route.ts"
import { remote } from '@/puck-remote';

// Preview links of other themes (needs `preview` in the config).
export const { GET } = remote.createPreviewHandler();
```

<Step>

### Wrap the Next config

```ts title="next.config.ts"
import { withPuckRemote } from '@puck-remote/next/config';

export default withPuckRemote({
  allowedDevOrigins: ['*.localhost', '127.0.0.1'], // dev hostnames: admin.localhost, and 127.0.0.1 for the editor
});
```

</Step>
<Step>

### Publish a theme

Create a theme (see [Build a theme](/docs/guides/theme-development)) and tell the CLI how to reach
your app's artifacts API, with the token your `authorize` accepts:

```ts title="my-theme/puck-remote.cli.ts"
import { defineCliConfig } from '@puck-remote/cli/config';

export default defineCliConfig({
  url: 'http://admin.localhost:3000/api/themes',
  headers: { authorization: `Bearer ${process.env.DEPLOY_TOKEN}` },
});
```

```bash
npx puck-remote push --live
```

</Step>
<Step>

### Point the editor's hostname at the app

The `app/theme/editor` page above renders the editor on `origins.editor`. Point that hostname (another
site) at this deployment. To customize the editor, see
[Customizing the editor](/docs/guides/editor-app).

</Step>
<Step>

### Run it

```bash
NODE_OPTIONS=--no-node-snapshot npx next dev
```

Open the site on `http://localhost:3000` and the admin page on `http://admin.localhost:3000/admin`
(with `origins.host` set to that origin and `origins.editor` to `http://127.0.0.1:3000` in
development). The editor must run on a different
origin from the admin pages (see [Origins](/docs/concepts/origins)).

</Step>
</Steps>
